2.1 Страны сбора

Для Cardiobacterium hominis страна берётся из метаданных. Если в поле несколько значений через двоеточие, остаётся первое. Пустые и неопределённые помечаются как Not provided.

import pandas as pd

missing_country = {"", "not provided", "not determined", "missing", "nan"}

def normalize_country(raw):
    country = (raw or "").strip()
    if country.lower() in missing_country:
        return "Not provided"
    return country.split(":")[0].strip()

df = pd.read_csv("../data/C_H/assembly_metadata/Cardiobacterium_hominis.tsv", sep="\t")
df["Country"] = df["Country"].fillna("Not provided").map(normalize_country)

result = (
    df.groupby("Country")
      .size()
      .reset_index(name="Count")
      .sort_values("Count", ascending=False)
)

result.to_csv("../results/tables/CH_countries.csv", index=False)

2.2 Датасет стран для iTOL

Для iTOL каждому листу дерева задаётся цвет страны. Четыре древних консенсуса идут одной серой полосой Ancient, современные — по стране из таблицы выше. Файл: data/C_H/old_iq_tree_no_outgroup/country_meta_itol.txt.

Code 
import csv
from collections import Counter
from pathlib import Path

metadata_path = Path("../data/C_H/assembly_metadata/Cardiobacterium_hominis.tsv")
output_path = Path("../data/C_H/old_iq_tree_no_outgroup/country_meta_itol.txt")

# Цвета стран
country_colors = {
    "USA": "#ff0000",
    "Germany": "#00ff00",
    "Australia": "#800080",
    "Denmark": "#0000ff",
    "Not provided": "#ffffff",
    "China": "#ffff00",
    "Brazil": "#ff69b4",
    "Spain": "#ffa500",
    "Papua New Guinea": "#87ceeb",
    "Hong Kong": "#d6d614",
    "Ancient": "#808080",
    "Switzerland": "#00ced1",
    "Cameroon": "#8b4513",
}

missing_country = {"", "not provided", "not determined", "missing", "nan"}


def normalize_country(raw):
    country = (raw or "").strip()
    if country.lower() in missing_country:
        return "Not provided"
    return country.split(":")[0].strip()

# 000008104510_consensus > BTC_Pereyaslavl-Zalessky_Russia_476_BP
# 000008107130_consensus > BTC_Komsomolets_Kurgan1_Stavropol_3149_BP
# 000008103140_consensus > BTC_Progress2_Caucasus_Kurgan1_6012_BP
# 000008101780_consensus > BTC_Ken_Su_Kyrgyzstan_2170_BP

ancient_samples = [
    "BTC_Pereyaslavl-Zalessky_Russia_476_BP*",
    "BTC_Komsomolets_Kurgan1_Stavropol_3149_BP*",
    "BTC_Progress2_Caucasus_Kurgan1_6012_BP*",
    "BTC_Ken_Su_Kyrgyzstan_2170_BP*",
]

ancient_color = country_colors["Ancient"]

rows = []
with metadata_path.open() as handle:
    for row in csv.DictReader(handle, delimiter="\t"):
        acc = row["AssemblyAccession"].strip()
        country = normalize_country(row.get("Country"))
        if country not in country_colors:
            raise KeyError(f"Нет цвета для страны {country!r} ({acc})")
        rows.append((acc, country_colors[country], country))

for name in ancient_samples:
    rows.append((name, ancient_color, "Ancient"))

ancient = [r for r in rows if r[2] == "Ancient"]
modern = [r for r in rows if r[2] != "Ancient"]
rows_sorted = ancient + modern

with output_path.open("w") as handle:
    handle.write(
        "DATASET_COLORSTRIP\n"
        "SEPARATOR TAB\n"
        "DATASET_LABEL\tCountry\n"
        "COLOR\t#ff0000\n"
        "STRIP_WIDTH\t25\n"
        "MARGIN\t5\n"
        "BORDER_WIDTH\t1\n"
        "BORDER_COLOR\t#000000\n"
        "\n"
        "DATA\n"
    )
    for name, color, country in rows_sorted:
        handle.write(f"{name}\t{color}\t{country}\n")

print(f"Записано {len(rows_sorted)} образцов в {output_path}")
print(f"  из них древних: {len(ancient)}")
print(Counter(country for _, _, country in rows_sorted))

2.4 Сборки C. hominis

Часть современных GFF уже собрана раньше. Сверяем их со свежим списком NCBI: заново аннотировать нужно только новые сборки, а в полный набор входят все текущие.

import pandas as pd
from os import listdir

Cardiobacterium_hominis = pd.read_csv('../data/C_H/assembly_metadata/Cardiobacterium_hominis.tsv', sep = '\t')

all_assembly = [name.split(".1")[0] + ".1" for name in Cardiobacterium_hominis["AssemblyAccession"]]
annotated_assembly = [name.split(".1")[0] + ".1" for name in listdir("../data/C_H/avaliable_gff_C_H")]

intersection = list(set(all_assembly) & set(annotated_assembly))
diff_all = list(set(all_assembly) - set(annotated_assembly))
diff_av = list(set(annotated_assembly) - set(all_assembly))

print(f'Текущее количество сборок в NCBI: {len(all_assembly)}')
